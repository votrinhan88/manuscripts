% ================================================================ 1. Motivation
\begin{frame}{1. Motivation --- the corpus bottleneck}
  \begin{itemize}
    \item Corpus cost recurs: every fine-tuning cycle, hyperparameter sweep, and continual-learning pass pays it again.
    \item Corpus size, not model size, is usually what binds.
  \end{itemize}
  \medskip
  \begin{block}{Dataset distillation \AIIGloss{(DD)}}
    Replace $\Dtrain$ with a surrogate $\Ddist \ll \Dtrain$ that trains a comparable model.\par\medskip
    \emph{What is the smallest textbook that teaches the same exam?}
  \end{block}
  \medskip
  \textbf{Text breaks image-space DD:} token decoding is non-differentiable, and embeddings are architecture-locked.
\end{frame}

  \smallskip
  \textbf{One gap cuts across all of them: uniform weighting.} At extreme compression, the budget goes to uninformative samples.
  \medskip
  \begin{block}{Three desiderata \AIIGloss{(all satisfied by TAKE)}}
    \textbf{(1) Knowledge-based weighting:} by task-aligned downstream contribution.\par
    \textbf{(2) Dataset-level objective:} the full training distribution, not batch by batch.\par
    \textbf{(3) Human-readable output:} auditable, and transferable across architectures.
  \end{block}
\end{frame}

% ================================================================ 2. Theory
\begin{frame}{2. Theory --- from DD to reweighted DM}
  The DD objective is bi-level, and intractable for discrete text:
  \[
    \min_{\Ddist}\; \mathbb{E}_{\theta_0 \sim p(\theta_0)} \left[ \mathbb{E}_{z \sim \Dtrain} \left[ \ell\!\left(z;\; F(\theta_0;\, \Ddist, \eta, T)\right) \right] \right]
  \]
  \textbf{Gradient-field equivalence:} a matching expected gradient field gives comparable parameters. Hence \emph{distribution matching} (DM):
  \[
    \min_{|\Ddist| = K}\; d\!\left(P_{\Ddist},\; P_{\Dtrain}\right)
  \]
  \textbf{But uniform matching follows density.} Budget goes to dense easy regions instead of decision boundaries. Reweighting by importance $w_n$ gives \emph{reweighted} DM (RDM):
  \[
    \min_{|\Ddist| = K}\; d\!\left(P_{\Ddist},\; P^w\right), \qquad P^w = \textstyle\sum_n w_n \delta_{z_n}
  \]
\end{frame}

\begin{frame}{2. Theory --- why the naive approach fails}
  Natural choice: the influence self-score $\mathcal{I}_\text{self} = \nabla_\theta \ell^\top H_{\hat{\theta}}^{-1} \nabla_\theta \ell$, task-aligned by construction. Set $w_n \propto \mathcal{I}_\text{self}$ and take top-$K$. \textbf{Two orthogonal failures:}
  \smallskip
  \begin{block}{Proposition 1: Hard-Sample Bias \AIIGloss{(the weights are biased)}}
    If clean samples have $\|\nabla_\theta \ell\| \leq \epsilon$ and noisy samples $\|\nabla_\theta \ell\| \geq \delta \gg \epsilon$ at convergence, then
    \[
      \Delta_\text{HSB} := \mathbb{E}_{\text{noisy}}[\mathcal{I}_\text{self}] - \mathbb{E}_{\text{clean}}[\mathcal{I}_\text{self}]
      \;\geq\; \frac{\delta^2 - \mathrm{cond}(H_{\hat{\theta}})\,\epsilon^2}{\lambda_{\max}(H_{\hat{\theta}})} \;>\; 0
    \]
    Clean gradients vanish at a local minimum by definition, so the bias is \textbf{structural}.
  \end{block}
  \begin{block}{Remark: Coverage \AIIGloss{(independent scoring $\neq$ distributional coverage)}}
    When two near-duplicates both score highly, top-$K$ takes both. A distributional objective keeps one and frees a slot for an uncovered region.
  \end{block}
\end{frame}

% ================================================================ 3. Method
\begin{frame}{3. Method --- TAKE at a glance}
  \small
  \begin{block}{Two open problems \AIIGloss{(carried over from the theory)}}
    \textbf{(a) Biased weighting.} Estimate $w_n$ by integrating gradient signals across the
    trajectory $\{\theta_t\}_{t=0}^{T-1}$, reducing $\Delta_\text{HSB}$. \hfill $\rightarrow$ Stage 1\par\smallskip
    \textbf{(b) Independent selection.} Solve the RDM objective directly for a metric $d$ that
    jointly optimises coverage of $P^w$. \hfill $\rightarrow$ Stage 3
  \end{block}

\begin{frame}{3. Method --- Stage 1: knowledge scores}
  \small
  The self-score needs Hessian inversion at convergence. We instead use per-sample gradient norms of a linear head $W$ over a frozen encoder $\phi$, column-normalized:
  \[
    I_{(n,t)} = \|\nabla_W \ell(z_n;\,\theta_t)\|,
    \qquad
    \kappa_n = \sum_{t=0}^{T-1} \frac{k(t)}{I_{(n,t)}},
    \qquad k(t) = e^{-\lambda t / T}
  \]
  \begin{itemize}
    \item Goodfellow identity $\nabla_W \ell = g_n \phi(x_n)^\top$ vectorizes the backward pass, with no per-sample loops.
    \item A well-fitted sample has a small norm, so the reciprocal inverts the ranking.
    \item A decreasing kernel $k$ concentrates mass before memorization sets in.
  \end{itemize}
  \begin{block}{Proposition 2: Trajectory Integration \AIIGloss{(reduces the HSB gap)}}
    For \emph{any} strictly decreasing $k$, $\kappa_n$ gives a strictly smaller noisy-over-clean gap than the last checkpoint.\quad Weights: $w_n = \kappa_n / \sum_{n'} \kappa_{n'}$.
  \end{block}
\end{frame}

\begin{frame}{3. Method --- Stages 2 \& 3: generate and distill}
  \begin{columns}[T,onlytextwidth]
    \column{.47\textwidth}
      \textbf{Stage 2: Synthetic pool}\par\medskip
      Direct subsampling risks privacy, and $K$ is too small to cover the support. Fine-tune a small LM on $\Dtrain$, then sample $\Dsyn$.\par\medskip
      $P_{\Dsyn} \neq P_{\Dtrain}$, and that helps:
      \begin{itemize}
        \item \textbf{Coverage expansion} of thin regions
        \item \textbf{Fluency bias} suppresses noise
        \item \textbf{Decoupling} from verbatim records
      \end{itemize}
    \column{.49\textwidth}
      \textbf{Stage 3: Discrete entropic OT}\par\medskip
      With $C_{nc} = \|\phi(z_n) - \phi(\tilde{z}_c)\|^2$, bare OT is $O(N^3)$. Entropic regularization is convex and Sinkhorn-solvable:
      \[
        \gamma^* = \arg\!\!\min_{\gamma \in \Pi(\mu,\nu)} \langle C, \gamma \rangle + \varepsilon \cdot \mathrm{KL}(\gamma \,\|\, \mu \otimes \nu)
      \]
      \begin{itemize}
        \item Soft coupling spreads mass $\Rightarrow$ diversity
        \item Take the $K$ candidates with most received mass $\sum_n \gamma^*_{nc}$
      \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{3. Method --- the theoretical chain}
  \begin{block}{Theorem: Gap Bound \AIIGloss{(distribution matching)}}
    With $\ell$ $L$-Lipschitz in the embedding metric and uniform stability $\beta_K = O(1/K)$,
    \[
      \bigl|\mathcal{L}(\Ddist) - \mathcal{L}(\Dtrain)\bigr|
      \;\leq\; 2L \cdot W_1(P^w,\,P_{\Ddist}) + \beta_K
    \]
    As $K \to \infty$, the bound is dominated by $W_1$, which the OT objective controls directly.
  \end{block}
  \vfill
  End-to-end guarantees:
  \begin{itemize}
    \item HSB is structural and unavoidable at a single checkpoint \hfill (Proposition 1)
    \item Trajectory integration strictly reduces it \hfill (Proposition 2)
    \item Minimizing OT cost controls the downstream loss gap under $P^w$ \hfill (Theorem)
    \item The residual gap from $P^w$ to $P_{\Dtrain}$ is bounded, computable from $\kappa_n$ \hfill (Remark)
  \end{itemize}
\end{frame}

\begin{frame}{4. Results --- experimental setup}
  \begin{columns}[T,onlytextwidth]
    \column{.48\textwidth}
      \textbf{Tasks and data}\par\medskip
      \begin{itemize}
        \item \textbf{Classification:} AG News (120K), IMDb (25K), SST-2 (67K)
        \item \textbf{NLI:} MNLI-m (393K), QNLI (105K), QQP (364K)
        \item Budgets: \textbf{0.1\%} and \textbf{20 samples/class}
        \item Accuracy, mean$_{\pm\text{s.e.}}$ over five runs
      \end{itemize}
      \medskip
      \textbf{Baselines:} Random; EDA (10$\times$ augmentation); DiLM (gradient matching $+$ $k$-centre); DaLLME (inverted embeddings).
    \column{.48\textwidth}
      \textbf{Implementation}\par\medskip
      \begin{itemize}
        \item $T = 20$ checkpoints, exponential kernel $\lambda^* = T/T_m$
        \item Encoder $\phi$: all-MiniLM-L6-v2 (frozen)
        \item Pool: Gemma-3 (270M) fine-tuned, $|\Dsyn| = 5{,}000$
        \item Sinkhorn: $\varepsilon = 0.05$, 200 iterations
        \item Single NVIDIA V100 (40\,GB)
      \end{itemize}
      \medskip
      \textbf{Downstream models:} LR (TF-IDF), TextCNN, TextRNN (GloVe), BERT; Siamese LR and BERT for NLI.
  \end{columns}
\end{frame}

\begin{frame}{5. Limitations \& ethical considerations}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{.48\textwidth}
      \textbf{Limitations}\par\medskip
      \begin{itemize}
        \item \textbf{Gradient-norm proxy.} Not Fisher-preconditioned; K-FAC / EKFAC are natural extensions.
        \item \textbf{Encoder dependence.} The OT cost inherits the geometry of $\phi$; $L$-Lipschitzness is only approximate for discrete text.
        \item \textbf{Pool coverage.} Bounded by the generator; fine-tuning on $\Dtrain$ mitigates this.
      \end{itemize}
    \column{.48\textwidth}
      \textbf{Ethical considerations}\par\medskip
      \begin{itemize}
        \item \textbf{Bias propagation.} Distillation concentrates $\Dtrain$'s statistics, demographic biases included, so audit \emph{before} distilling.
        \item \textbf{Privacy risk.} Fine-tuned LMs can memorise. Screen $\Dsyn$ by distance-to-closest-record; use DP fine-tuning in regulated domains.
      \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{6. Takeaways}
  \begin{itemize}
    \item The \textbf{hard-sample bias} is the core failure mode of gradient-based text DD: at convergence, noisy samples dominate single-checkpoint influence scores.
    \medskip
    \item \textbf{TAKE} corrects it: gradient norms integrated along the trajectory give knowledge scores $\kappa_n$, which weight a discrete OT objective selecting a compact, human-readable corpus.
    \medskip
    \item Guarantees end to end: HSB is structural, trajectory integration strictly reduces it, and the loss gap is bounded by the OT cost.
    \medskip
    \item Empirically: best across six benchmarks at 0.1\% / 20 per class, and it transfers across backbones.
  \end{itemize}
\end{frame}
